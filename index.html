const GITHUB_BRANCH = "main";



/* 이미지가 들어가는 기본 폴더 */


const IMAGE_ROOT = "images";

/* =========================================================

   GITHUB API

========================================================= */


async function githubAPI(path) {


    const url =

        `https://api.github.com/repos/` +

        `${GITHUB_OWNER}/${GITHUB_REPO}/contents/` +

        `${path}?ref=${GITHUB_BRANCH}`;


    const response =

        await fetch(url);


    if (!response.ok) {


        throw new Error(

            "GitHub API Error: " + path

        );


    }


    return await response.json();


}



/* =========================================================

   RAW IMAGE URL

========================================================= */


function imageURL(path) {


    return (

        `https://raw.githubusercontent.com/` +

        `${GITHUB_OWNER}/` +

        `${GITHUB_REPO}/` +

        `${GITHUB_BRANCH}/` +

        `${path}`

    );


}



/* =========================================================

   폴더 안의 작업 목록 가져오기

========================================================= */


async function getWorks(category) {


    try {


        const items =

            await githubAPI(

                `${IMAGE_ROOT}/${category}`

            );



        return items


            .filter(

                item =>

                    item.type === "dir"

            )


            .sort(

                (a, b) =>

                    a.name.localeCompare(

                        b.name,

                        "ko"

                    )

            );


    }


    catch (error) {


        console.log(

            "카테고리 없음:",

            category

        );


        return [];


    }


}



/* =========================================================

   작업 폴더 안의 사진 가져오기

========================================================= */


async function getImages(path) {


    try {


        const items =

            await githubAPI(path);



        return items


            .filter(


                item =>


                    item.type === "file" &&


                    /\.(jpg|jpeg|png|webp|gif)$/i

                        .test(item.name)


            )


            .sort(


                (a, b) =>


                    a.name.localeCompare(

                        b.name,

                        undefined,

                        {

                            numeric: true,

                            sensitivity: "base"

                        }

                    )


            );


    }


    catch (error) {


        return [];


    }


}



/* =========================================================

   전체 사이트 데이터 읽기

========================================================= */


async function loadSiteData() {


    for (

        const category of CATEGORIES

    ) {


        const works =

            await getWorks(category);



        siteData[category] = [];



        for (

            const work of works

        ) {


            const images =

                await getImages(

                    work.path

                );



            siteData[category].push({


                name: work.name,


                path: work.path,


                images: images


            });


        }


    }


}

function renderHome() {


    hero.innerHTML = "";



    const allWorks = [];



    CATEGORIES.forEach(


        category => {


            siteData[category]

                .forEach(


                    work => {


                        if (

                            work.images.length > 0

                        ) {


                            allWorks.push({


                                category:

                                    category,


                                work:

                                    work


                            });


                        }


                    }


                );


        }


    );



    /*

       첫 번째 3개 작업을 메인에 표시


       나중에 순서를 바꾸고 싶으면

       폴더 이름 앞에 숫자를 붙이는 방식으로

       관리할 수도 있음.

    */



    const selected =

        allWorks.slice(0, 3);



    if (

        selected.length === 0

    ) {


        for (

            let i = 0;

            i < 3;

            i++

        ) {


            const card =

                document.createElement(

                    "article"

                </div>


            `;



            hero.appendChild(card);


        }


        return;


    }



    selected.forEach(


        item => {


            const card =

                document.createElement(

                    "article"

                );


            card.className =

                "hero-card";



            const image =

                document.createElement(

                    "img"

                );


            image.className =

                "hero-image";

                );

                item.work.name;

                document.createElement(

                    "div"

                );


            caption.className =

                "hero-caption";



            caption.textContent =


                `${capitalize(item.category)}, ` +

                `${item.work.name}`;

            );

        }


    );


}

function capitalize(text) {


    if (!text) {

        return "";

    }


    return (

        text.charAt(0).toUpperCase() +

        text.slice(1)

    );


}
