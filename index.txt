const GITHUB_OWNER =
    "songnansae";


const GITHUB_REPO =
    "SONGNANSAE";


const GITHUB_BRANCH =
    "main";


const IMAGE_ROOT =
    "images";


/*
    메뉴에 들어가는 카테고리
*/

    "commercial",
    "magazine",
    "personal",
    "film"

*/

    commercial: [],
    magazine: [],
    personal: [],
    film: []

};

    메인 3개 카드의 현재 프로젝트

const currentIndex = {

    commercial: 0,
    magazine: 0,
    personal: 0

/* ==================================================
   GITHUB API URL
================================================== */

function githubContentsURL(path) {


    const encodedPath =

        path
            .split("/")
            .map(
                part =>
                    encodeURIComponent(part)
            )
            .join("/");


    return (

        "https://api.github.com/repos/" +

        GITHUB_OWNER +

        "/" +

        GITHUB_REPO +

        "/contents/" +

        encodedPath +

        "?ref=" +

        encodeURIComponent(
            GITHUB_BRANCH
        )

    );

}


/* ==================================================
   NATURAL SORT
================================================== */

function naturalSort(a, b) {


    return a.localeCompare(

        b,

        undefined,

        {

            numeric: true,

            sensitivity: "base"

        }

    );

}


/* ==================================================
   GITHUB CONTENTS
================================================== */

async function getGitHubContents(path) {


    try {


        const response =

            await fetch(

                githubContentsURL(
                    path
                )

            );


        if (!response.ok) {

            console.log(
                "GitHub API error:",
                path,
                response.status
            );

            return [];

        }


        const data =
            await response.json();


        if (!Array.isArray(data)) {

            return [];

        }


        return data;


    }

    catch (error) {


        console.log(
            "GitHub fetch error:",
            error
        );


        return [];

    }

}


/* ==================================================
   IMAGE CHECK
================================================== */

function isImage(file) {


    if (
        !file ||
        !file.name
    ) {

        return false;

    }


    const extension =

        file.name
            .split(".")
            .pop()
            .toLowerCase();


    return [

        "jpg",
        "jpeg",
        "png",
        "webp",
        "gif"

    ].includes(
        extension
    );

}


/* ==================================================
   LOAD HOME BANNER
================================================== */

async function loadHomeBanner() {


    const banner =
        document.getElementById(
            "homeBanner"
        );


    banner.innerHTML =
        "";


    /*
        images/home/

        안의 사진만 가져옴.

        폴더는 무시하고
        실제 이미지 파일만 사용.
    */

    const files =
        await getGitHubContents(
            "images/home"
        );


    const imageFiles =

        files

            .filter(
                file =>
                    file.type === "file" &&
                    isImage(file)
            )

            .sort(
                (a, b) =>
                    naturalSort(
                        a.name,
                        b.name
                    )
            );


    /*
        사진이 없으면
        아무것도 표시하지 않음.
    */

    if (
        imageFiles.length === 0
    ) {

        return;

    }


    /*
        각 사진을
        원본 비율 그대로 표시.
    */

    imageFiles.forEach(
        file => {


            const img =
                document.createElement(
                    "img"
                );


            img.className =
                "home-banner-item";


            img.src =
                file.download_url;


            img.alt =
                "";


            img.loading =
                "eager";


            banner.appendChild(
                img
            );


        }
    );

}


/* ==================================================
   LOAD CATEGORY
================================================== */

async function loadCategory(
    category
) {


    const categoryPath =

        `${IMAGE_ROOT}/${category}`;


    const contents =

        await getGitHubContents(
            categoryPath
        );


    /*
        category 안의
        실제 폴더만 프로젝트.
    */

    const folders =

        contents

            .filter(
                item =>
                    item.type === "dir"
            )

            .sort(
                (a, b) =>
                    naturalSort(
                        a.name,
                        b.name
                    )
            );


    const loadedProjects = [];


    for (
        const folder of folders
    ) {


        const folderPath =

            `${categoryPath}/${folder.name}`;


        const files =

            await getGitHubContents(
                folderPath
            );


        const imageFiles =

            files

                .filter(
                    file =>
                        file.type === "file" &&
                        isImage(file)
                )

                .sort(
                    (a, b) =>
                        naturalSort(
                            a.name,
                            b.name
                        )
                );


        /*
            사진 없는 폴더는 제외.
        */

        if (
            imageFiles.length === 0
        ) {

            continue;

        }


        loadedProjects.push({

            name:
                folder.name,

            category:
                category,

            images:

                imageFiles.map(
                    file => ({

                        name:
                            file.name,

                        url:
                            file.download_url

                    })
                )

        });


    }


    projects[category] =
        loadedProjects;

}

   LOAD EVERYTHING
================================================== */

async function loadAllProjects() {


    /*
        홈 배너와
        4개 카테고리를
        동시에 불러옴.
    */

    await Promise.all([

        loadHomeBanner(),

        ...categories.map(
            category =>

                    category
                )
        )

    ]);


    renderHome();

    renderProjectList();

}

/* ==================================================
   RENDER HOME CARD
================================================== */

function renderHomeCard(

    category,

    imageContainerId,

    captionId

) {


    const imageContainer =

        document.getElementById(
            imageContainerId
        );


    const caption =

        document.getElementById(
            captionId
        );


    imageContainer.innerHTML =
        "";


    caption.innerHTML =
        "";


    const categoryProjects =
        projects[category];


    /*
        프로젝트가 없는 경우
    */

    if (

        !categoryProjects ||

        categoryProjects.length === 0

    ) {


        imageContainer.innerHTML =

            '<div class="empty-card"></div>';


        caption.innerHTML = `

            <span
                class="home-caption-category"
            >
                ${capitalize(category)}
            </span>

        `;


        return;

    }


    /*
        현재 프로젝트
    */

    const index =

        currentIndex[category] %

        categoryProjects.length;


    const project =

        categoryProjects[index];


    /*
        첫 번째 파일이
        메인 대표 이미지.

        따라서
        01_ 파일을 넣으면
        그 사진이 메인.
    */

    const cover =
        project.images[0];


    if (!cover) {

        return;

    }


    const img =
        document.createElement(
            "img"
        );


    img.className =
        "home-image loading";


    img.src =
        cover.url;


    img.alt =
        project.name;


    img.onload = () => {

        img.classList.remove(
            "loading"
        );

    };


    img.onerror = () => {

        imageContainer.innerHTML =

            '<div class="empty-card"></div>';

    };


    imageContainer.appendChild(
        img
    );


    /*
        캡션
    */

    caption.innerHTML = `

        <span
            class="home-caption-category"
        >
            ${capitalize(category)}
        </span>

        <span
            class="home-caption-project"
        >
            ${escapeHTML(
                project.name
            )}
        </span>

    `;

}


/* ==================================================
   NEXT PROJECT
================================================== */

function nextProject(
    category
) {


    if (

        !projects[category] ||

        projects[category].length === 0

    ) {

        return;

    }


    currentIndex[category]++;


    if (

        currentIndex[category] >=

        projects[category].length

    ) {

        currentIndex[category] =
            0;

    }


    renderHomeCard(

        category,

        `${category}ImageContainer`,

        `${category}Caption`

    );

}


/* ==================================================
   MENU CATEGORY
================================================== */

function selectMenuCategory(
    category
) {


    selectedMenuCategory =
        category;


    document
        .querySelectorAll(
            ".menu-tab"
        )
        .forEach(
            tab => {


                if (

                    tab.dataset.category ===
                    category

                ) {

                    tab.classList.add(
                        "active"
                    );

                }

                else {

                    tab.classList.remove(
                        "active"
                    );

                }

            }
        );


    renderProjectList();

}


/* ==================================================

    const container =

            "projectList"
        );


    container.innerHTML =
        "";


    const categoryProjects =

        projects[
            selectedMenuCategory
        ];


    if (

        !categoryProjects ||

        categoryProjects.length === 0

    ) {

        return;

    }


    categoryProjects.forEach(

        (project, index) => {


            const item =
                document.createElement(
                    "div"
                );


            item.className =
                "project-list-item";


            item.textContent =
                project.name;


            item.onclick = () => {

                openProject(

                    selectedMenuCategory,

                    index

                );

            };


            container.appendChild(
                item
            );


        }

    );

}


/* ==================================================

================================================== */

    category,

    index

) {

        projects[category][index];


    if (!project) {

        return;

    }


    document.getElementById(
        "home"
    ).style.display =
        "none";


    document.getElementById(
        "detail"
    ).classList.add(
        "active"
    );


    document.getElementById(
        "detailTitle"
    ).textContent =
        project.name;


    document.getElementById(
        "detailCategory"
    ).textContent =
        capitalize(
            category
        );


    const detailImages =
        document.getElementById(
            "detailImages"
        );


    detailImages.innerHTML =
        "";


    project.images.forEach(

        image => {


            const img =
                document.createElement(
                    "img"
                );


            img.src =
                image.url;


            img.alt =
                project.name;


            img.loading =
                "lazy";


            detailImages.appendChild(
                img
            );


        }

    );


    closeMenu();


    window.scrollTo(
        0,
        0
    );

}

/* ==================================================
   ESCAPE HTML
================================================== */

function escapeHTML(
    value
) {


    return String(value)

        .replace(
            /&/g,
            "&amp;"
        )

        .replace(
            /</g,
            "&lt;"
        )

        .replace(
            />/g,
            "&gt;"
        )

        .replace(
            /"/g,
            "&quot;"
        )

        .replace(
            /'/g,
            "&#039;"
        );

}


/* ==================================================
   CAPITALIZE
================================================== */

function capitalize(
    value
) {


    return (

        value.charAt(0).toUpperCase() +

        value.slice(1)

    );

}
